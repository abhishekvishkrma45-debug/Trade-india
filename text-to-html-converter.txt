							<script> 
    let balance = 1000; 
    let selectedColour = null; 
    let betAmount = 100; 
    let timeLeft = 10; 
 
    function updateTimer() { 
        document.getElementById('timer').innerText = timeLeft; 
        if (timeLeft === 0) { 
            calculateResult(); 
            timeLeft = 10; 
        } else { 
            timeLeft--; 
        } 
    } 
 
    setInterval(updateTimer, 1000); 
 
    function placeBet(colour) { 
        if (balance < betAmount) { 
            document.getElementById('status').innerText = "Insufficent Balance!"; 
            document.getElementById('status').style.color = "red"; 
            return; 
        } 
        selectedColour = colour; 
        document.getElementById('status').innerText = "Bet Placed on: " + colour; 
        document.getElementById('status').style.color = "#ffeb3b"; 
    } 
 
    function calculateResult() { 
        let randomIndex = Math.floor(Math.random() * 10); 
        let winningColour = ''; 
         
        if (randomIndex === 0) winningColour = 'Violet'; 
        else if (randomIndex % 2 === 0) winningColour = 'Green'; 
        else winningColour = 'Red'; 
 
        let statusText = "Result: " + winningColour; 
 
        if (selectedColour !== null) { 
            if (selectedColour === winningColour) { 
                let winMultiplier = winningColour === 'Violet' ? 4.5 : 2; 
                let winAmt = betAmount * winMultiplier; 
                balance += winAmt; 
                statusText += " | Aap Jeet Gaye ₹" + winAmt + "!"; 
            } else { 
                balance -= betAmount; 
                statusText += " | Aap Haar Gaye ₹" + betAmount; 
            } 
        } 
 
        document.getElementById('balance').innerText = balance; 
        document.getElementById('status').innerText = statusText; 
        document.getElementById('status').style.color = "white"; 
 
        let historyList = document.getElementById('history-list'); 
        let item = document.createElement('div'); 
        item.className = 'history-item'; 
        item.innerHTML = `
								<span>Result: 
									<b>${winningColour}</b>
								</span>`; 
        historyList.prepend(item); 
 
        selectedColour = null; 
    }
